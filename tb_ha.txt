`timescale 1ns / 1ps

module half_adder_tb;

    // Testbench signals
    reg  a;
    reg  b;
    wire sum;
    wire carry;

    // Instantiate Device Under Test (DUT)
    half_adder dut (
        .a(a),
        .b(b),
        .sum(sum),
        .carry(carry)
    );

    initial begin
        // Monitor signals in console
        $monitor("Time = %0t | A = %b, B = %b | Sum = %b, Carry = %b", $time, a, b, sum, carry);

        // Apply all 4 input combinations
        a = 0; b = 0;
        #10;

        a = 0; b = 1;
        #10;

        a = 1; b = 0;
        #10;

        a = 1; b = 1;
        #10;

        $finish; // End simulation
    end

endmodule
